type Point=readonly [number,number];
type Place={label:string;point:Point};
type RoadRoute={distance:number;duration:number;points:Point[]};
function isPoint(p:unknown):p is Point{return Array.isArray(p)&&p.length===2&&p.every(x=>typeof x==="number"&&Number.isFinite(x))&&Math.abs(p[0])<=180&&Math.abs(p[1])<=90;}
function obj(x:unknown):Record<string,unknown>|null{return x!==null&&typeof x==="object"&&!Array.isArray(x)?x as Record<string,unknown>:null;}
export function parseGeoapifyPlaces(input:unknown):Place[]{const r=obj(input)?.["results"];if(!Array.isArray(r))return [];return r.slice(0,8).flatMap(v=>{const p=obj(v);if(!p||p["country_code"]!=="mz"||typeof p["formatted"]!=="string")return [];const point=[p["lon"],p["lat"]];return isPoint(point)&&p["formatted"].trim()?[{label:p["formatted"].slice(0,240),point}]:[];});}
export function parseGeoapifyRoute(input:unknown):RoadRoute|null{const root=obj(input);if(root?.["type"]!=="FeatureCollection"||!Array.isArray(root["features"]))return null;const f=obj(root["features"][0]);const p=obj(f?.["properties"]),g=obj(f?.["geometry"]);const distance=p?.["distance"],duration=p?.["time"];const lines=g?.["coordinates"];if(typeof distance!=="number"||typeof duration!=="number"||!Number.isFinite(distance)||!Number.isFinite(duration)||distance<=0||duration<=0||g?.["type"]!=="MultiLineString"||!Array.isArray(lines)||lines.length!==1)return null;const points:unknown=lines[0];if(!Array.isArray(points)||points.length<2||points.length>20000||!points.every(isPoint))return null;return {distance,duration,points:points as Point[]};}
export function geoapifyMode(vehicle:"car"|"motorcycle"){return vehicle==="car"?"drive":"motorcycle";}

export async function geoSearch(query:string,key:string){const u=geoUrl("geocode/search",key);u.searchParams.set("text",query);u.searchParams.set("format","json");u.searchParams.set("filter","circle:35.2459,-13.2999,20000");u.searchParams.set("bias","proximity:"+launchCity.center.join(","));u.searchParams.set("lang","pt");u.searchParams.set("limit","8");return parseGeoapifyPlaces(await request(u));}

async function reverse(point:Point,key:string){const u=geoUrl("geocode/reverse",key);u.searchParams.set("lon",String(point[0]));u.searchParams.set("lat",String(point[1]));u.searchParams.set("format","json");u.searchParams.set("lang","pt");const v=obj(await request(u));const rows=v?.["results"];const r=Array.isArray(rows)?obj(rows[0]):null;if(!r||r["country_code"]!=="mz")return null;const parts=[r["suburb"]??r["district"]??r["neighbourhood"],r["city"]??r["town"]??r["village"]].filter((x):x is string=>typeof x==="string"&&x.trim().length>0);const label=[...new Set(parts)].join(", ")||(typeof r["formatted"]==="string"?r["formatted"]:"");return label.trim()?{label:label.slice(0,160),point}:null;}

export async function geoTiles(z:number,tiles:TileCoordinate[]):Promise<{status:"unconfigured"|"ok"|"unavailable";tiles:{x:number;y:number;src:string}[]}>{
 const key=geoKey();if(!key)return {status:"unconfigured",tiles:[]};
 if(tiles.length<1||tiles.length>9||tiles.some(p=>!validTile(z,p))||new Set(tiles.map(p=>p.x+"/"+p.y)).size!==tiles.length)return {status:"unavailable",tiles:[]};
 const results=await Promise.all(tiles.map(async p=>{try{const u=new URL("https://maps.geoapify.com/v1/tile/osm-carto/"+z+"/"+p.x+"/"+p.y+".png");u.searchParams.set("apiKey",key);const r=await fetch(u,{redirect:"error",signal:AbortSignal.timeout(10000),headers:{Accept:"image/png"}});if(!r.ok||r.headers.get("content-type")?.split(";")[0]!=="image/png")return null;const bytes=new Uint8Array(await r.arrayBuffer());if(bytes.length<8||bytes.length>400000||![137,80,78,71,13,10,26,10].every((b,i)=>bytes[i]===b))return null;let raw="";for(let i=0;i<bytes.length;i+=8192)raw+=String.fromCharCode(...bytes.subarray(i,i+8192));return {...p,src:"data:image/png;base64,"+btoa(raw)};}catch{return null;}}));
 const images=results.filter((p):p is NonNullable<typeof p>=>p!==null);return {status:images.length===tiles.length?"ok":"unavailable",tiles:images};
}
